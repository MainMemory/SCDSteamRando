sub ObjectMain
	switch Object.State
	case 0
		Object.XPos=Screen.XSize
		Object.Value3=0
		Object.Value0=0
		Object.Value1=0
		Object.InkEffect=2
		Object.Alpha=0
		switch Engine.Language
		case 0
		case 5
			Object.PropertyValue=12
			break
		case 1
			Object.PropertyValue=15
			break
		case 2
		case 3
		case 4
			Object.PropertyValue=13
			break

		endswitch
		if Options.GameMode!=-2
			if Player.Lives<1
				Object.Value6=1
			else
				if Stage.ActiveList==SPECIAL_STAGE
					Object.Value6=0
				else
					Object.Value6=0
				endif
			endif
		else
			Object.Value6=1
		endif
		if Options.GameMode==0
			Object.Value7=1
		endif
		if Options.GameMode==2
			Object.Value7=1
		endif
		Object.State++
		break
	case 1
		Object.Value3=Object.Value0
		Object.Value3*=Screen.XSize
		Object.Value3/=12
		Object.Value4=Object.Value0
		Object.Value4*=240
		Object.Value4/=12
		TempValue0=Object.Value0
		TempValue0<<=7
		TempValue0/=12
		Object.XPos=Screen.XSize
		Object.XPos-=TempValue0
		if Object.Value0<Object.PropertyValue
			Object.Value0++
		else
			Object.Value2=Object.XPos
			Object.Value2+=48
			Object.Value0=0
			Object.State++
		endif
		break
	case 2
		if Object.Value0<256
			Object.Value0+=16
			if Object.Value0<255
				Object.Alpha=Object.Value0
			else
				Object.Alpha=255
			endif
		else
			Object.Value0=0
			Object.Alpha=255
			Object.State++
		endif
		break
	case 3
			if KeyPress.Up==1
				PlaySfx(23,0)
				Object.Value0=0
				Object.AnimationTimer=0
				Object.Frame=0
				Object.Value2=Screen.XSize
				Object.Value1--
				if Object.Value1<0
					Object.Value1=3
				endif
				if Object.Value6==1
					if Object.Value1==1
						Object.Value1--
					endif
				endif
				if Object.Value7==1
					if Object.Value1==2
						Object.Value1--
					endif
				endif
			endif
			if KeyPress.Down==1
				PlaySfx(23,0)
				Object.Value0=0
				Object.AnimationTimer=0
				Object.Frame=0
				Object.Value2=Screen.XSize
				Object.Value1++
				if Object.Value1>3
					Object.Value1=0
				endif
				if Object.Value6==1
					if Object.Value1==1
						Object.Value1++
					endif
				endif
				if Object.Value7==1
					if Object.Value1==2
						Object.Value1++
					endif
				endif
			endif
			if KeyPress.Start==1
				PlaySfx(27,0)
				Object.State=4
				Object.Alpha=248
				Object.Value0=0
			endif
			if KeyPress.ButtonA==1
				PlaySfx(27,0)
				Object.State=4
				Object.Alpha=248
				Object.Value0=0
			endif
			if KeyPress.ButtonC==1
				PlaySfx(27,0)
				Object.State=4
				Object.Alpha=248
				Object.Value0=0
			endif
		if Object.Value0<60
			Object.Value0++
		else
			Object.Frame=Object.AnimationTimer
			Object.Frame>>=4
			Object.Frame++
			Object.AnimationTimer++
			Object.AnimationTimer&=31
		endif
		break
	case 4
		Object.Value0++
		Object.Value0&=3
		Object.Value2+=4
		if Object.Alpha>0
			Object.Alpha-=8
		else
			if Object.Value1==0
				Object.Value2=Object.XPos
				Object.Value2+=48
				Object.Value4=202
				Object.Value0=0
				Object.Value1=128
				Object.State=5
				Stage.State=STAGE_RUNNING
			else
				Object.Value2=Object.XPos
				Object.Value2+=48
				StopMusic()
				Object.State=6
			endif
		endif
		break
	case 5
		if Object.Value1>0
			if KeyPress.Start==1
				KeyPress.Start=0
				Stage.State=STAGE_PAUSED
				PauseMusic()
				PlaySfx(27,0)
				StopSfx(14)
				StopSfx(24)
				StopSfx(25)
				Object.XPos+=128
				Object.Value2=0
				Object.Value4=-24
				Object.Value5=0
				Object.State=0
			endif
			Object.Value1-=8
			Object.XPos+=16
			Object.Value2+=16
			Object.Value4+=16
		else
			ResumeMusic()
			ResetObjectEntity(Object.EntityNo,TypeName[BlankObject],0,0,0)
			if Engine.FrameSkipSetting>0
				Engine.FrameSkipTimer=0
			endif
			TileLayer[0].Type=4
		endif
		break
	case 6
		if Object.XPos>-32
			Object.XPos-=16
			Object.Value2+=16
		else
			if Options.GameMode==1
				ArrayPos0=Options.SaveSlot
				ArrayPos0*=800
				ArrayPos0+=800
				TempValue2=Stage.Seconds
				TempValue2*=100
				TempValue3=Stage.Minutes
				TempValue3*=6000
				TempValue2+=TempValue3
				TempValue2+=Stage.MilliSeconds
				SaveRAM[ArrayPos0]+=TempValue2
				ArrayPos0++
				TempValue3=SaveRAM[ArrayPos0]
				SaveRAM[ArrayPos0]++
				ArrayPos0+=4
				TempValue3*=3
				ArrayPos0+=TempValue3
				SaveRAM[ArrayPos0]=Stage.ListPos
				SaveRAM[ArrayPos0]+=80
				ArrayPos0++
				SaveRAM[ArrayPos0]=84
				ArrayPos0++
				SaveRAM[ArrayPos0]=TempValue2
				WriteSaveRAM()
			endif
			switch Object.Value1
			case 1
				LampPost.Check=0
				LoadStage()
				break
			case 2
				Stage.TimeEnabled=0
				if Options.GameMode==1
					ArrayPos1=Options.SaveSlot
				else
					ArrayPos1=Options.NoSaveSlot
				endif
				ArrayPos1<<=3
				ArrayPos1++
				SaveRAM[ArrayPos1]=3
				ArrayPos1++
				SaveRAM[ArrayPos1]=0
				ArrayPos1++
				//REPLACE1
				ArrayPos1++
				SaveRAM[ArrayPos1]=0
				ArrayPos1++
				SaveRAM[ArrayPos1]=0
				ArrayPos1++
				SaveRAM[ArrayPos1]=50000
				ArrayPos1++
				SaveRAM[ArrayPos1]=0
				if Options.GameMode==1
					ArrayPos1=Options.SaveSlot
				else
					ArrayPos1=Options.NoSaveSlot
				endif
				ArrayPos1*=800
				ArrayPos1+=800
				SaveRAM[ArrayPos1]=0
				ArrayPos1++
				SaveRAM[ArrayPos1]=0
				ArrayPos1++
				SaveRAM[ArrayPos1]=0
				ArrayPos1++
				SaveRAM[ArrayPos1]=0
				ArrayPos1++
				SaveRAM[ArrayPos1]=0
				Player.Lives=3
				Player.Score=0
				Player.ScoreBonus=50000
				Good_Future_List=0
				Good_Future_Count=0
				Good_Future=0
				Transporter_Destroyed=0
				MetalSonic_Destroyed=0
				MetalSonic_List=0
				SpecialStage.TimeStones=0
				SpecialStage.ListPos=0
				LampPost.Check=0
				MetalSonicKidnappedAmy=0
				Options.SaveComplete=1
				if Options.GameMode!=-2
					Stage.ActiveList=REGULAR_STAGE
				else
					Stage.ActiveList=SPECIAL_STAGE
				endif
				//REPLACE2
				LoadStage()
				if Options.GameMode!=-2
					if Stage.PlayerListPos==0
						if SaveRAM[361]==0
							SaveRAM[38]=0
						else
							SaveRAM[38]+=1
						endif
						Options.Soundtrack=SaveRAM[38]
						LoadVideo("Opening")
						if SaveRAM[361]==0
							SaveRAM[38]=0
						else
							SaveRAM[38]-=1
						endif
					endif
					if Stage.PlayerListPos==2
						LoadVideo("Amy")
					endif
					SetScreenFade(0,0,0,Object.Value0)
				endif
				break
			case 3
				Good_Future_Count=0
				Good_Future_List=0
				Good_Future=0
				Transporter_Destroyed=0
				MetalSonic_Destroyed=0
				Stage.ActiveList=PRESENTATION_STAGE
				LampPost.Check=0
				if Options.GameMode==2
					Stage.ListPos=2
					TimeAttack.Result=0
				else
					Stage.ListPos=11
				endif
				LoadStage()
				break
			endswitch
		endif
		break
	endswitch
endsub


sub ObjectDraw
		switch Object.State
		case 0
		case 1
		case 2
			if Object.Value5<240
				if Object.Value4>Object.Value5
					TempValue0=Object.Value4
					TempValue0-=Object.Value5
					DrawRect(0,Object.Value5,426,TempValue0,255,255,255,128)
					Object.Value5=Object.Value4
				endif
			endif
			break
		case 3
		case 4
			//Empty
			break
		case 5
			if Object.Alpha<128
				DrawRect(0,0,Screen.XSize,240,255,255,255,Object.Alpha)
			endif
			break
		case 6
			TempValue0=Screen.XSize
			TempValue0-=96
			DrawRect(TempValue0,0,96,240,0,0,0,255)
			break
		endswitch

			DrawSpriteScreenXY(3,Object.XPos,0)
			DrawSpriteScreenXY(3,Object.XPos,32)
			DrawSpriteScreenXY(3,Object.XPos,64)
			DrawSpriteScreenXY(3,Object.XPos,96)
			DrawSpriteScreenXY(3,Object.XPos,128)
			DrawSpriteScreenXY(3,Object.XPos,160)
			DrawSpriteScreenXY(3,Object.XPos,192)
			DrawSpriteScreenXY(3,Object.XPos,224)
			TempValue0=Object.XPos
			TempValue0+=128

		switch Object.State
		case 0
		case 1
		case 2
		case 3
		case 4
			DrawRect(TempValue0,0,128,Screen.YSize,0,0,0,255)
			DrawRect(0,202,Object.Value3,13,0,0,0,255)
			DrawSpriteScreenXY(4,0,202)
			TempValue0=Object.XPos
			TempValue0+=48

			if Object.State != 4
				if Object.Value2>TempValue0
					TempValue1=TempValue0
					TempValue1-=Object.Value2
					TempValue1>>=3
					Object.Value2+=TempValue1
				endif
			endif

			TempValue1=48
			break
		case 5
			DrawRect(TempValue0,0,128,Screen.YSize,0,0,0,255)
			DrawRect(0,Object.Value4,Screen.XSize,13,0,0,0,255)
			DrawSpriteScreenXY(4,0,Object.Value4)
			break
		case 6
			DrawRect(TempValue0,0,128,Screen.YSize,0,0,0,255)
			break
		endswitch


	switch Object.State
	case 0
	case 1
	case 2
	case 3
		DrawSpriteScreenFX(6,FX_INK,TempValue0,TempValue1)

		if Object.Value1==0
			DrawSpriteScreenFX(5,FX_INK,Object.Value2,TempValue1)
			DrawSpriteScreenFX(Object.Frame,FX_INK,TempValue0,TempValue1)
		endif
		TempValue1+=48

		if Object.Value6==0
			DrawSpriteScreenFX(7,FX_INK,TempValue0,TempValue1)
		else	   
			DrawSpriteScreenFX(10,FX_INK,TempValue0,TempValue1)  
		endif

		if Object.Value1==1
			DrawSpriteScreenFX(5,FX_INK,Object.Value2,TempValue1)
			DrawSpriteScreenFX(Object.Frame,FX_INK,TempValue0,TempValue1)
		endif
		TempValue1+=48

		if Object.Value7==0
			DrawSpriteScreenFX(9,FX_INK,TempValue0,TempValue1)
		else	   
			DrawSpriteScreenFX(11,FX_INK,TempValue0,TempValue1)  
		endif
		
		if Object.Value1==2
			DrawSpriteScreenFX(5,FX_INK,Object.Value2,TempValue1)
			DrawSpriteScreenFX(Object.Frame,FX_INK,TempValue0,TempValue1)
		endif
		TempValue1+=48

		DrawSpriteScreenFX(8,FX_INK,TempValue0,TempValue1)
		if Object.Value1==3
			DrawSpriteScreenFX(5,FX_INK,Object.Value2,TempValue1)
			DrawSpriteScreenFX(Object.Frame,FX_INK,TempValue0,TempValue1)
		endif
		break
	case 4
		if Object.Value1==0
			if Object.Value0<2
				DrawSpriteScreenXY(6,TempValue0,TempValue1)
			endif
			DrawSpriteScreenXY(5,Object.Value2,TempValue1)
			DrawSpriteScreenXY(Object.Frame,TempValue0,TempValue1)
		else
			DrawSpriteScreenFX(6,FX_INK,TempValue0,TempValue1)
		endif
		TempValue1+=48
		if Object.Value1==1
			if Object.Value0<2
				DrawSpriteScreenXY(7,TempValue0,TempValue1)
			endif
			DrawSpriteScreenXY(5,Object.Value2,TempValue1)
			DrawSpriteScreenXY(Object.Frame,TempValue0,TempValue1)
		else
			if Object.Value6==0
				DrawSpriteScreenFX(7,FX_INK,TempValue0,TempValue1)
			else
				DrawSpriteScreenFX(10,FX_INK,TempValue0,TempValue1)		   
			endif
		endif
		TempValue1+=48
		if Object.Value1==2
			if Object.Value0<2
				DrawSpriteScreenXY(9,TempValue0,TempValue1)
			endif
			DrawSpriteScreenXY(5,Object.Value2,TempValue1)
			DrawSpriteScreenXY(Object.Frame,TempValue0,TempValue1)
		else
			if Object.Value7==0
				DrawSpriteScreenFX(9,FX_INK,TempValue0,TempValue1)
			else	   
				DrawSpriteScreenFX(11,FX_INK,TempValue0,TempValue1)  
			endif
		endif
		TempValue1+=48
		if Object.Value1==3
			if Object.Value0<2
				DrawSpriteScreenXY(8,TempValue0,TempValue1)
			endif
			DrawSpriteScreenXY(5,Object.Value2,TempValue1)
			DrawSpriteScreenXY(Object.Frame,TempValue0,TempValue1)
		else
			DrawSpriteScreenFX(8,FX_INK,TempValue0,TempValue1)
		endif
		break
	case 5
	case 6
		switch Object.Value1
		case 0
			DrawSpriteScreenXY(6,Object.Value2,48)
			DrawSpriteScreenXY(Object.Frame,Object.Value2,48)
			break
		case 1
			DrawSpriteScreenXY(6,Object.Value2,96)
			DrawSpriteScreenXY(Object.Frame,Object.Value2,96)
			break
		case 2
			if Object.Value7==0
				DrawSpriteScreenFX(9,FX_INK,TempValue0,TempValue1)
			else	   
				DrawSpriteScreenFX(11,FX_INK,TempValue0,TempValue1)  
			endif
			DrawSpriteScreenXY(Object.Frame,Object.Value2,144)
			break
		case 3
			if Object.Value7==0
				DrawSpriteScreenFX(9,FX_INK,TempValue0,TempValue1)
			else	   
				DrawSpriteScreenFX(11,FX_INK,TempValue0,TempValue1)  
			endif
			DrawSpriteScreenXY(Object.Frame,Object.Value2,192)
			break

		endswitch
		break

	endswitch
endsub
